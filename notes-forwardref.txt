use forwardref and useImperativeHandle when passing refs to another components.

Sample usage:
App.jsx
import Form from './Form';
import React from "react";

export function App() {
  const formRef = React.useRef();
  
  function handleRestart() {
      if (formRef.current) {
        // using useImperativeHandle function makes the method exposed on the parent component
        formRef.current.clear();
      }
  }

  return (
    <div id="app">
      <button onClick={handleRestart}>Restart</button>
      <Form ref={formRef} />
    </div>
  );
}

Form.jsx
import React from "react";

const Form = React.forwardRef((props, ref) => {
    const formRef = React.useRef();
    
    // using this React function makes the clear method exposed to the parent that is passing the ref.
    // note: that useImperativeHandle should always returns an object.
    React.useImperativeHandle(ref, () => {
        return {
            clear: () => {
                if (formRef.current) {
                    formRef.current.reset();
                }
            }
        }
    })
    
    return (
    <form ref={formRef}>
      <p>
        <label>Name</label>
        <input type="text" />
      </p>

      <p>
        <label>Email</label>
        <input type="email" />
      </p>
      <p id="actions">
        <button>Save</button>
      </p>
    </form>
  );
})

export default Form;
